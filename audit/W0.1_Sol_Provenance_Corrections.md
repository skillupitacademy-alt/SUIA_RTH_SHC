## Verification Commands

```powershell
# Validate CSV parses correctly
$csv = Import-Csv "audit/Audit_Finding_Root_Cause_Crosswalk.csv"
Write-Host "Total rows: $($csv.Count)"

# Check all status values are PLANNED
$csv | Group-Object status | Select-Object Count, Name

# Verify provenance key uniqueness
$duplicates = $csv | Group-Object source_document,source_section_or_line,original_finding_id |
    Where-Object { $_.Count -gt 1 }
if ($duplicates) {
    Write-Host "❌ Duplicate provenance keys found"
    $duplicates | Format-Table Count, Name -AutoSize
} else {
    Write-Host "✅ All 91 provenance keys are unique"
}

# Verify no broad "Section: X" locators remain
$broadLocators = $csv | Where-Object {
    $_.'source_section_or_line' -match '^Section: ' -and
    $_.'source_section_or_line' -notmatch 'Line \d+'
}
if ($broadLocators) {
    Write-Host "❌ Broad locators still present:"
    $broadLocators | Format-Table original_finding_id, source_section_or_line -AutoSize
} else {
    Write-Host "✅ All 91 entries have exact line numbers"
}

# Sample line number verification
$sampleFindings = @(
    @{ID='SHC-TAXONOMY-CONTRACT-003'; ExpectedLine=5291}
    @{ID='SU-AUTH-024'; ExpectedLine=18106}
    @{ID='PH05-005'; ExpectedLine=34772}
    @{ID='RTH-R1-001'; ExpectedLine=29027}
)

foreach ($sample in $sampleFindings) {
    $entry = $csv | Where-Object { $_.'original_finding_id' -eq $sample.ID }
    if ($entry.'source_section_or_line' -match $sample.ExpectedLine) {
        Write-Host "✅ $($sample.ID) resolves to Line $($sample.ExpectedLine)"
    } else {
        Write-Host "❌ $($sample.ID) line number mismatch"
    }
}
```
